% =====================================================================
%  ELAJ — Supplementary Information
%
%  Build:  bash build_supp.sh
%
%  SEPARATE DOCUMENT, SEPARATE NUMBERING. Nature ships the Supplementary
%  Information as its own PDF with its own figure, table and reference
%  numbering, prefixed so a reference from the main text is unambiguous.
%  Every NMI supplementary read for DISCUSSION_REGISTER.md does this and
%  carries a cover page the journal generates at proof, which is why
%  there is no title block here.
%
%  THE SECTION ORDER IS TAKEN FROM TIGON AND MIDAS, the two NMI
%  supplementaries whose structure was read page by page:
%      contents → Notes → Methods → Tables → Figures → References
%  MIDAS puts Notes before Tables and Figures; TIGON puts Figures first.
%  Either is accepted. Notes first is chosen because our Notes carry the
%  arguments and the tables and figures are what they cite.
%
%  WHAT A NOTE IS. A short prose section making ONE argument, with its
%  own figures and tables where it needs them. Not an appendix of raw
%  output. See SUPPLEMENTARY_STRUCTURE.md.
% =====================================================================

\documentclass[11pt, a4paper]{article}

\input{../preamble}

% ── Supplementary numbering ────────────────────────────────────────────
% Prefixed so "Supplementary Fig. 3" can never be read as "Fig. 3".
\renewcommand{\thefigure}{S\arabic{figure}}
\renewcommand{\thetable}{S\arabic{table}}
\renewcommand{\thesection}{Supplementary Note \arabic{section}}
\renewcommand{\thesubsection}{\arabic{section}.\arabic{subsection}}

% SUBSECTIONS ARE UNNUMBERED BUT STILL LISTED. Across the four NMI
% supplementaries read for this, numbered subsections appear only where a
% Note is a reference list of definitions (scGPT's Note 12, metric
% calculations). Where a Note makes an argument, TIGON and MIDAS both
% subdivide with unnumbered headings. secnumdepth stops the numbering at
% the Note; tocdepth keeps the headings in the contents.
\setcounter{secnumdepth}{1}
\setcounter{tocdepth}{2}

% The contents line reserves a narrow box for a section number, and
% "Supplementary Note 1" overflows it by 104 pt. Widen the box rather than
% shortening the label: the label is what makes a cross-reference from the
% main text unambiguous.
\makeatletter
\renewcommand*\l@section[2]{\@dottedtocline{1}{0em}{10.5em}{#1}{#2}}
\renewcommand*\l@subsection[2]{\@dottedtocline{2}{10.5em}{2.5em}{#1}{#2}}
\makeatother

\title{\vspace{-2em}\textbf{Supplementary Information}\\[0.3em]
\large ELAJ: a single-cell disease atlas for endometriosis}
\date{}

\begin{document}
\maketitle
\vspace{-2em}

\tableofcontents

% EVERY FIGURE AND TABLE BY TITLE, not just the prose headings. Three of the
% four comparators do this: TIGON lists every item by number and title,
% Eye2Gene lists every figure with its legend, MIDAS lists both levels. A
% reader looking for the ablation UMAPs scans for that phrase, not for
% "Supplementary Note 5". Short captions, \caption[short]{long}, are what
% these lists print, so every float needs one.
\listoffigures
\listoftables
\newpage

% ── Supplementary Notes ───────────────────────────────────────────────
% \clearpage after every Note so its floats land inside it rather than

%

% 6 can be folded in to note 3", so the label transfer is now a limitation
% of the claims axis inside Note 3 with one table and one figure, and "Note
% 8 can be folded in to note 7", so the marker-reproducibility test is a
% subsection of the claims note. File names track the Note numbers.
\input{notes/N1-gene-grouping}\clearpage

\input{notes/N4-frontier-arms}\clearpage

% After the Notes, as MIDAS orders it (AW, 15 Nov). Unnumbered, because
% \thesection prints "Supplementary Note N"; the main text points at its
% subsections by name, e.g. (Supplementary Methods, \emph{Severity}).
\input{methods/supplementary_methods}\clearpage

% ── References ───────────────────────────────────────────────────────
% The supplementary carries its OWN reference list, numbered separately
% from the manuscript's. MIDAS and TIGON both do this: a reader of the
% supplementary alone must be able to resolve every citation in it. Same
% .bib as the manuscript, so an entry cannot drift between the two.
\renewcommand{\refname}{Supplementary references}
\bibliography{../references}
